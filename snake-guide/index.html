snake_final.py

This uses only turtle, time, and random from Python.

import turtle
import time
import random

STEP = 20
WALL_LIMIT = 280
START_POSITIONS = [(0, 0), (-20, 0), (-40, 0)]


class Snake:
    def __init__(self):
        self.segments = []
        self.create_snake()
        self.head = self.segments[0]

    def create_snake(self):
        for index, position in enumerate(START_POSITIONS):
            color = "green" if index == 0 else "grey"
            self.add_segment(position, color)

    def add_segment(self, position, color="grey"):
        segment = turtle.Turtle("square")
        segment.color(color)
        segment.penup()
        segment.goto(position)
        self.segments.append(segment)

    def extend(self):
        last_position = self.segments[-1].position()
        self.add_segment(last_position)

    def move(self):
        for number in range(len(self.segments) - 1, 0, -1):
            x = self.segments[number - 1].xcor()
            y = self.segments[number - 1].ycor()
            self.segments[number].goto(x, y)
        self.head.forward(STEP)

    def up(self):
        if self.head.heading() != 270:
            self.head.setheading(90)

    def down(self):
        if self.head.heading() != 90:
            self.head.setheading(270)

    def left(self):
        if self.head.heading() != 0:
            self.head.setheading(180)

    def right(self):
        if self.head.heading() != 180:
            self.head.setheading(0)


class Food(turtle.Turtle):
    def __init__(self):
        super().__init__()
        self.shape("circle")
        self.color("deepskyblue")
        self.penup()
        self.shapesize(stretch_wid=0.5, stretch_len=0.5)
        self.speed("fastest")
        self.refresh()

    def refresh(self):
        random_x = random.randint(-13, 13) * STEP
        random_y = random.randint(-13, 13) * STEP
        self.goto(random_x, random_y)


class Scoreboard(turtle.Turtle):
    def __init__(self):
        super().__init__()
        self.score = 0
        self.color("white")
        self.penup()
        self.hideturtle()
        self.goto(0, 255)
        self.update_score()

    def update_score(self):
        self.clear()
        self.write(
            f"Score: {self.score}",
            align="center",
            font=("Arial", 22, "normal"),
        )

    def add_point(self):
        self.score += 10
        self.update_score()

    def game_over(self):
        self.goto(0, 0)
        self.color("tomato")
        self.write(
            "Game Over",
            align="center",
            font=("Arial", 34, "bold"),
        )


# 1. Game screen
screen = turtle.Screen()
screen.setup(width=600, height=600)
screen.bgcolor("black")
screen.title("Snake Game")
screen.tracer(0)

# 2. Game objects
snake = Snake()
food = Food()
scoreboard = Scoreboard()

# 3. Keyboard controls
screen.listen()
screen.onkey(snake.up, "Up")
screen.onkey(snake.down, "Down")
screen.onkey(snake.left, "Left")
screen.onkey(snake.right, "Right")

# 4. Game loop
DELAY_START = 0.16
delay = DELAY_START
game_is_on = True

while game_is_on:
    screen.update()
    time.sleep(delay)
    snake.move()

    # Food collision
    if snake.head.distance(food) < 15:
        food.refresh()
        snake.extend()
        scoreboard.add_point()
        delay = max(0.06, delay - 0.01)

    # Wall crash
    hit_wall = (
        abs(snake.head.xcor()) > WALL_LIMIT
        or abs(snake.head.ycor()) > WALL_LIMIT
    )
    if hit_wall:
        game_is_on = False
        scoreboard.game_over()

    # Tail crash
    for segment in snake.segments[1:]:
        if snake.head.distance(segment) < 10:
            game_is_on = False
            scoreboard.game_over()
            break

screen.mainloop()

Find a Bug by What You See

Look for your problem below. Then check the matching line of code.

06
The window opens, then stops

Check for import time. Make sure the lines under while game_is_on: have the same indent.

The body piles up in one spot

Check the last -1 in range(..., 0, -1). Make sure the tail moves first.

The arrow keys do nothing

Put screen.listen() before onkey. Click the game window once, then try again.

The snake eats but does not grow

Make sure snake.extend() is indented inside the food if block.

Score numbers pile up

Make sure self.clear() is the first line in update_score().

A quick U-turn ends the game

Compare the four if self.head.heading() != ... checks in the direction functions.

Game Over does not appear

Check that you call scoreboard.game_over(). Make sure screen.mainloop() comes after the loop.
